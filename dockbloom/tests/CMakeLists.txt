cmake_minimum_required(VERSION 3.21)
project(DockBloomQtTests LANGUAGES CXX)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
add_executable(dockbloom-ui-tests ui-tests.cpp ../src/workspace-config.cpp ../src/workspace.cpp ../src/workspace-ui.cpp)
target_include_directories(dockbloom-ui-tests PRIVATE ../src)
target_compile_features(dockbloom-ui-tests PRIVATE cxx_std_17)
target_link_libraries(dockbloom-ui-tests PRIVATE Qt6::Core Qt6::Widgets)
if(MSVC)
  target_compile_options(dockbloom-ui-tests PRIVATE /W4)
else()
  target_compile_options(dockbloom-ui-tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME dockbloom-ui-tests COMMAND dockbloom-ui-tests)
set_tests_properties(dockbloom-ui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_FRONTEND_API "Use obs-frontend-api" OFF)
option(ENABLE_QT "Use Qt" OFF)
include(compilerconfig)
include(defaults)
include(helpers)
add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin.cpp)
find_package(libobs REQUIRED)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc objdetect dnn)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs ${OpenCV_LIBS})
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE ${OpenCV_INCLUDE_DIRS})
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
